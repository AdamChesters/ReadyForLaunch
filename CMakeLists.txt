cmake_minimum_required(VERSION 3.25)
project(ReadyForLaunch VERSION 0.2.0 LANGUAGES CXX RC)
set(RFL_VERSION "0.2.0-alpha.1")
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
configure_file(src/resources.rc.in generated/resources.rc @ONLY)
if(NOT WIN32)
  message(FATAL_ERROR "ReadyForLaunch currently targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(launch_core src/model.cpp src/engine.cpp)
target_include_directories(launch_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(launch_core PUBLIC nlohmann_json::nlohmann_json)
add_library(launch_windows src/windows.cpp src/updates.cpp)
target_link_libraries(launch_windows PUBLIC launch_core ole32 shell32 shlwapi propsys advapi32 comdlg32 dwmapi winhttp bcrypt)
add_executable(ReadyForLaunch WIN32 src/main.cpp src/ui.cpp src/images.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/resources.rc")
target_link_libraries(ReadyForLaunch PRIVATE launch_windows imgui::imgui d3d11 dxgi d3dcompiler windowscodecs)
enable_testing()
add_executable(launch_tests tests/tests.cpp)
target_link_libraries(launch_tests PRIVATE launch_core)
add_test(NAME scheduler_and_profiles COMMAND launch_tests)
add_executable(launch_fixture WIN32 tests/fixture.cpp)
add_executable(windows_tests tests/windows_tests.cpp)
target_link_libraries(windows_tests PRIVATE launch_windows)
add_test(NAME windows_process_lifecycle COMMAND windows_tests "$<TARGET_FILE:launch_fixture>")
add_executable(update_tests tests/update_tests.cpp)
target_link_libraries(update_tests PRIVATE launch_windows)
add_test(NAME update_verification COMMAND update_tests)
add_executable(ui_tests tests/ui_tests.cpp src/ui.cpp src/images.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/resources.rc")
target_link_libraries(ui_tests PRIVATE launch_windows imgui::imgui d3d11 dxgi d3dcompiler windowscodecs)
add_test(NAME profile_ui_interactions COMMAND ui_tests)
install(TARGETS ReadyForLaunch RUNTIME DESTINATION .)
install(FILES README.md CHANGELOG.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(DIRECTORY assets DESTINATION .)
